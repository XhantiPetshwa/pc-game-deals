<script>
// ---------- Source data (Part A-C) ----------
const deals = [
  { id: 1, title: "Hollow Knight", normalPrice: 14.99, salePrice: 7.49, store: "Steam" },
  { id: 2, title: "Stardew Valley", normalPrice: 13.99, salePrice: 13.99, store: "GOG" },
  { id: 3, title: "Celeste", normalPrice: 19.99, salePrice: 4.99, store: "Steam" },
  { id: 4, title: "Hades", normalPrice: 24.99, salePrice: 12.49, store: "Epic Games" },
  { id: 5, title: "Portal 2", normalPrice: 9.99, salePrice: 0.99, store: "Steam" },
  { id: 6, title: "Terraria", normalPrice: 9.99, salePrice: 9.99, store: "GOG" },
  { id: 7, title: "Disco Elysium", normalPrice: 39.99, salePrice: 9.99, store: "GOG" },
  { id: 8, title: "Cuphead", normalPrice: 19.99, salePrice: 14.99, store: "Steam" },
];

// ---------- Part A: small functions ----------
const isOnSale = (game) => game.salePrice < game.normalPrice;

const getOnSaleGames = (games) => games.filter(isOnSale);

function discountPercent(game) {
  if (!game.normalPrice || game.normalPrice <= 0) return 0;
  if (!isOnSale(game)) return 0;
  return Math.round((1 - game.salePrice / game.normalPrice) * 100);
}

function findCheapestGame(games) {
  if (!games.length) return null;
  return games.reduce((a, b) => (b.salePrice < a.salePrice ? b : a));
}

function searchGames(games, query) {
  const q = query.trim().toLowerCase();
  if (!q) return games;
  return games.filter((g) => g.title.toLowerCase().includes(q));
}

// Quick sanity checks (open DevTools console to see them):
console.log("On sale:", getOnSaleGames(deals).map(g => g.title));
console.log("Celeste discount %:", discountPercent(deals[2])); // 75
console.log("Cheapest:", findCheapestGame(deals).title); // Portal 2
console.log("Search 'hades':", searchGames(deals, "HADES").map(g => g.title));

// ---------- Part B + C: render + live search/filter ----------
let currentDeals = deals;
const listEl = document.getElementById("deals");
const statusEl = document.getElementById("status");

function render() {
  const query = document.getElementById("search").value;
  const onlySale = document.getElementById("onSaleOnly").checked;
  let shown = searchGames(currentDeals, query);
  if (onlySale) shown = getOnSaleGames(shown);

  statusEl.textContent = `${shown.length} deal(s) shown. Cheapest shown: ${findCheapestGame(shown)?.title ?? "—"}`;

  listEl.innerHTML = shown.map((g) => `
    <article class="card">
      <h2>${g.title} ${isOnSale(g) ? `<span class="tag">ON SALE</span>` : ""}</h2>
      <p class="muted">${g.store}</p>
      <p>
        ${isOnSale(g) ? `<span class="old">R${g.normalPrice.toFixed(2)}</span>` : ""}
        <span class="price">R${g.salePrice.toFixed(2)}</span>
        ${isOnSale(g) ? `— ${discountPercent(g)}% off` : "— not discounted"}
      </p>
    </article>
  `).join("") || "<p>No matches.</p>";
}

document.getElementById("search").addEventListener("input", render);
document.getElementById("onSaleOnly").addEventListener("change", render);

// ---------- Part C stretch: CheapShark (USD in API, shown as-is) ----------
document.getElementById("loadLive").addEventListener("click", async () => {
  statusEl.textContent = "Loading live deals from CheapShark...";
  try {
    const res = await fetch("https://www.cheapshark.com/api/1.0/deals?storeID=1&upperPrice=15&pageSize=8");
    if (!res.ok) throw new Error("API failed");
    const data = await res.json();
    currentDeals = data.map((d, i) => ({
      id: 1000 + i,
      title: d.title,
      normalPrice: Number(d.normalPrice),
      salePrice: Number(d.salePrice),
      store: "CheapShark / Steam",
    }));
    render();
  } catch (e) {
    statusEl.textContent = "Could not load live deals (network/CORS). Showing the supplied deals instead.";
    currentDeals = deals;
    render();
  }
});

render();
</script>
